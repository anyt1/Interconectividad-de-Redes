Algoritmo_STP():
	Para c ada switch S en la red:
		Bridge_ID(S) = Prioridad + Direccion_MAC_local(S)
		Root_Bridge = S // Inicialmente cada switch se cree raíz a sí mismo
		Root_Path_Cost = 0
		Root_Port = NULO


ciclos
	Mientras la red este activa:
		Enviar BPDU (Bridge Protocol Data Unit) con:
			- Mi_Bridge_ID
			- Root_Bridge_Conocido
			- Rooth_Path_Cost

		Para cada BPDU recibida de un vecino V:
			Si (Root_Bridge_vecino tiene mejor Bridge_ID que mi Root_Bridge actual) 0
				(Tienen el mismo Root_Bridge y Costo_Vecino + Costo_enlace < mi_Root_Path_Cost):

				Root_Bridge = Root_Bridge_vecino
				Root_Path_Cost = Costo_vecino + Costo_enlace
				Root_Port = Puerto_hacia_vecino(V)

		// Asignación de roles de puertos
		Para cada puerto P del switch S:
			Si P == Root_Port:
				Estado(P) = FORWARDING (Puerto Raiz)
			Sino si P es el puerto designado para el segmento (menor costo hacia la raíz):
				Estado(P) = FORWARDING (Puerto designado)
			Sino:
				Estado(P) = BLOCKING (Puerto bloqueado para evitar bucles)
